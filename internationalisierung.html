    7. Internationalisierung

    7.1. Beispiel 21: Eine zweisprachige Anwendung

    Eine zweisprachige Anwendung hat drei Arten von Texten, die übersetzt werden müssen:

    • Anwendungstexte: Beschriftungen und Meldungen. Sie werden in Wörterbüchern gespeichert, eines pro Sprache;
    • Datumsangaben und Beträge: Das Objekt [Intl] von [JavaScript] formatiert sie in der gewünschten Sprache (siehe Beispiel 09);
    • Texte von [Ionic] selbst: die Monate und Tage seines Kalenders, seine Schaltflächen. Der Kalender ([<ion-datetime>]) erhält die Sprache über sein Attribut [locale].

    // =========================================================================
// src/app/ex21-i18n/translations.ts - Wörterbücher, eine aktuelle Sprache
// -------------------------------------------------------------------------
// Jede Sprache hat ihr WÖRTERBUCH, eine JSON-Datei, die mit
// der Anwendung ausgeliefert wird (public/i18n/ex21-fr.json, ex21-en.json), in der jeder Text
// einen SCHLÜSSEL hat ("HOME.HELLO"):
//   - t('HOME.HELLO', { name: 'Marie' }): der Text der AKTUELLEN
//     Sprache, in dem die Parameter {{name}} ersetzt werden;
//   - t('HOME.APPOINTMENTS', { count: n }): ein Text im Plural; wir
//     wählen den Schlüssel APPOINTMENTS_zero, _one oder _other nach n und den Regeln
//     der Sprache (im Französischen sind 0 und 1 Singular; im Englischen
//     nur 1);
//   - changeLanguage('fr'): lädt das andere Wörterbuch. Die Sprache und das
//     Wörterbuch sind SIGNALS: Die Templates, die t() aufrufen, werden
//     aktualisiert, sie übersetzen sich neu.
// Das sind die Konventionen der JavaScript-Bibliothek i18next: Die
// Wörterbücher sind die der Clients in Vue.js, Angular und React von
// RdvMedecins ... und der Server, der nur SCHLÜSSEL zurückgibt
// ("ERRORS.LOGIN_TAKEN"), verwendet sie ebenfalls.
// =========================================================================

import { signal } from '@angular/core';
import { Preferences as DeviceStorage } from '@capacitor/preferences';

export const languages = ['fr', 'en'];

// ein Wörterbuch: Texte oder andere Wörterbücher (verschachteltes JSON)
type Dictionary = { [key: string]: string | Dictionary };

export class Translations {
  private readonly _language = signal('en');
  private readonly _dictionary = signal<Dictionary>({});
  readonly language = this._language.asReadonly();

  // prefix: der Anfang des Namens der Wörterbücher (public/i18n/<prefix>-<language>.json)
  constructor(private readonly prefix: string) {}

  // die Startsprache: die auf dem Gerät gespeicherte, sonst die des
  // Telefons (falls bekannt), sonst Englisch
  async start(): Promise<void> {
    const stored = (await DeviceStorage.get({ key: `${this.prefix}.language` })).value;
    const device = navigator.language.substring(0, 2); // "en-US" -> "en"
    await this.changeLanguage(stored ?? (languages.includes(device) ? device : 'en'));
  }

  async changeLanguage(language: string): Promise<void> {
    // fetch: liest eine mit der Anwendung ausgelieferte Datei (Ordner public/)
    const response = await fetch(`i18n/${this.prefix}-${language}.json`);
    this._dictionary.set((await response.json()) as Dictionary);
    this._language.set(language);
    // die Sprache der Seite (Screenreader, Silbentrennung ...)
    document.documentElement.lang = language;
    await DeviceStorage.set({ key: `${this.prefix}.language`, value: language });
  }

  // "HOME.HELLO" -> dictionary['HOME']['HELLO']; undefined, falls nicht vorhanden
  private find(key: string): string | undefined {
    let node: string | Dictionary | undefined = this._dictionary();
    for (const part of key.split('.')) {
      node = typeof node === 'object' ? node[part] : undefined;
    }
    return typeof node === 'string' ? node : undefined;
  }

  t(key: string, params: Record<string, string | number> = {}): string {
    let text: string | undefined;
    const count = params['count'];
    if (typeof count === 'number') {
      // Intl.PluralRules: die Pluralform ("one", "other" ...), nach den
      // Regeln der Sprache; der Schlüssel _zero hat, sofern vorhanden, bei 0 Vorrang
      const form = new Intl.PluralRules(this._language()).select(count);
      text = (count === 0 ? this.find(`${key}_zero`) : undefined) ?? this.find(`${key}_${form}`);
    }
    text ??= this.find(key);
    // ein unbekannter Schlüssel wird unverändert angezeigt: So sehen wir sofort, was fehlt
    if (text === undefined) return key;
    // {{name}} -> der Wert des Parameters name
    return text.replace(/\{\{(\w+)\}\}/g, (all, name: string) => String(params[name] ?? all));
  }
}
    • Zeile 25: die Sprachen der Anwendung;
    • Zeile 28: der Typ eines Wörterbuchs – ein Objekt, bei dem jede Eigenschaft entweder Text oder ein weiteres Wörterbuch ist (der Typ ist rekursiv: Er verweist auf sich selbst);
    • Zeilen 30–33: Die Übersetzungen speichern die aktuelle Sprache und ihr Wörterbuch in Variablen. Eine einfache Klasse, kein Service [@Injectable]: Jeder Nutzer erstellt sie mit seinem eigenen Präfix (Zeile 36; Beispiel 25 hat seine eigenen Wörterbücher, [ex25-fr.json] ...). Sie wird der Seite von einer „Factory“ bereitgestellt (wird später behandelt);
    • Zeile 36: [private readonly prefix] in den Parametern des Konstruktors deklariert UND initialisiert ein Feld: eine Abkürzung von [TypeScript];
    • Zeilen 40–44: die Ausgangssprache: die auf dem Gerät gespeicherte (das Plugin [@capacitor/preferences] aus Beispiel 20), andernfalls die Sprache des Telefons ([navigator.language]: „en-US“, wovon die ersten beiden Buchstaben behalten werden), sofern bekannt; andernfalls Englisch;
    • Zeilen 46–54: Sprache wechseln: das Wörterbuch mit [fetch] lesen, der Funktion von [JavaScript] , die eine Adresse liest (hier eine mit der Anwendung ausgelieferte Datei) und ein [Promise] zurückgibt, das die Antwort darstellt; deren JSON-Inhalt dekodieren ([response.json()], ein weiteres [Promise]); das Wörterbuch und die Sprache in den jeweiligen Feldern speichern; die Sprache der Seite aktualisieren (das Attribut [lang] des Tags [<html>], das für Screenreader nützlich ist); die Auswahl auf dem Gerät speichern. Wir hätten den HTTP-Client von [Angular] (Beispiel 22) verwenden können: [fetch] vermeidet die Abhängigkeit von ihm in einer einfachen Klasse;
    • Zeilen 57–63: Suche nach einem Schlüssel „HOME.HELLO“ im Wörterbuch: Wir steigen Teil für Teil ([split('.')]) durch die verschachtelten JSON-Objekte hinab. [typeof node === 'object'] ist eine Typprüfung, die den Typ der Variablen in dem Zweig, in dem sie wahr ist, „eingrenzt“: [TypeScript] weiß dann, dass [node] ein Wörterbuch ist. Das Lesen des Signals ([this._dictionary()], Zeile 58) ist unerlässlich: Ein Template, das [t()] aufruft, liest das Signal auf diese Weise und wird aktualisiert, wenn sich das Wörterbuch ändert;
    • Zeilen 65–79: die Übersetzung. Bei einem Plural (ein numerischer Parameter [count]) liefert [Intl.PluralRules] die Pluralform gemäß den Regeln der Sprache: [select(n)] ergibt [one] oder [other] (andere Sprachen kennen auch [two], [few], [many]). Im Französischen sind 0 und 1 Singular ([one]); im Englischen nur 1. Der Schlüssel [_zero] hat, falls vorhanden, Vorrang vor 0 (Zeile 72), genau wie bei [i18next]. Ein unbekannter Schlüssel wird unverändert angezeigt (Zeile 76): So sieht man sofort, was fehlt. Schließlich (Zeile 78) wird jedes [{{name}}] durch den Wert des Parameters ersetzt: ein regulärer Ausdruck [/\{\{(\w+)\}\}/g] und eine Funktion, die jeden Treffer ([all]) und den in Klammern erfassten Namen ([name]) erhält.

    Warum nicht das offizielle [Angular]-Werkzeug verwenden? [Angular] hat ein eigenes Internationalisierungssystem ([@angular/localize]): Man markiert die Texte der Templates ([<p i18n>Hello</p>]), extrahiert die Übersetzungen in Dateien, und die Kompilierung erzeugt EINE Anwendung pro Sprache. Das ist die beste Lösung für eine Website, deren gesamte Texte im Voraus bekannt sind. Sie wechselt die Sprache jedoch nicht „im laufenden Betrieb“, und unser Server liefert Schlüssel („ERRORS.LOGIN_TAKEN“), die zur Laufzeit übersetzt werden müssen: Ein Wörterbuch, das über den Schlüssel abgefragt wird, eignet sich dafür weit besser und erlaubt es, Wörterbücher mit anderen Clients zu teilen. Es gibt dafür auch Bibliotheken für [Angular] ([ngx-translate], [Transloco]): Unser Service ist eine minimale Version davon.

    // =========================================================================
// src/app/ex21-i18n/ex21.page.ts - eine Anwendung in zwei Sprachen
// -------------------------------------------------------------------------
//   - die TEXTE der Anwendung: die Wörterbücher (translations.ts),
//     der Seite per Dependency Injection bereitgestellt (providers);
//   - die Datumsangaben und die Beträge: das Objekt Intl von JavaScript, in der
//     aktuellen Sprache (siehe Beispiel 09);
//   - die Texte von Ionic SELBST (der Kalender): das Attribut locale
//     von <ion-datetime> und seine Schaltflächen, übersetzt durch unsere Wörterbücher.
// Ein Sprachwechsel übersetzt den ganzen Bildschirm neu, ohne Neustart.
// =========================================================================

import { Component, computed, effect, inject, signal } from '@angular/core';
import { Title } from '@angular/platform-browser';
import {
  IonBackButton, IonButton, IonButtons, IonCard, IonCardContent, IonCardHeader, IonCardSubtitle, IonContent,
  IonDatetime, IonHeader, IonIcon, IonLabel, IonModal, IonSegment, IonSegmentButton, IonTitle, IonToolbar,
  SegmentValue,
} from '@ionic/angular';
import { addIcons } from 'ionicons';
import { calendarOutline } from 'ionicons/icons';
import { languages, Translations } from './translations';

// was der RdvMedecins-Server zurückgeben würde (siehe das Kapitel zur Blackbox)
const serverResponses: { statusCode: number; key: string; params: Record<string, string> }[] = [
  { statusCode: 409, key: 'ERRORS.LOGIN_TAKEN', params: {} },
  { statusCode: 429, key: 'ERRORS.TOO_MANY_ATTEMPTS', params: { minutes: '15' } },
];

@Component({
  selector: 'app-ex21',
  imports: [
    IonHeader, IonToolbar, IonButtons, IonBackButton, IonTitle, IonSegment, IonSegmentButton, IonLabel, IonContent,
    IonCard, IonCardHeader, IonCardSubtitle, IonCardContent, IonButton, IonIcon, IonModal, IonDatetime,
  ],
  templateUrl: './ex21.page.html',
  // ein für die Seite SPEZIFISCHER Provider: die Übersetzungen der Wörterbücher ex21-*.json
  providers: [{ provide: Translations, useFactory: () => new Translations('ex21') }],
  styles: `
    ion-segment { width: 110px; margin-inline-end: 8px; }
    ion-segment-button { min-width: 50px; }
    ion-modal.calendar { --width: fit-content; --height: fit-content; --border-radius: 8px; }
    .code { font-family: monospace; font-size: 12px; }
  `,
})
export class Ex21Page {
  readonly tr = inject(Translations);
  readonly languages = languages;
  readonly serverResponses = serverResponses;
  readonly count = signal(2);
  private readonly today = new Date();

  // Formatierung in der AKTUELLEN Sprache: wird neu berechnet, wenn sie sich ändert
  readonly date = computed(() =>
    new Intl.DateTimeFormat(this.tr.language(), { dateStyle: 'full' }).format(this.today),
  );
  readonly price = computed(() =>
    new Intl.NumberFormat(this.tr.language(), { style: 'currency', currency: 'EUR' }).format(1234.5),
  );
  // die Sprache von ion-datetime: ein vollständiges "Locale" (Sprache-Land)
  readonly locale = computed(() => (this.tr.language() === 'fr' ? 'fr-FR' : 'en-US'));

  constructor() {
    addIcons({ calendarOutline });
    this.tr.start();
    // der Titel des Dokuments (Browser-Tab), übersetzt: ein Effect,
    // der erneut ausgeführt wird, wenn sich die Sprache ändert
    const title = inject(Title);
    effect(() => title.setTitle(this.tr.t('APP.TITLE')));
  }

  // das Segment der oberen Leiste: Sein Wert (vom Typ SegmentValue: eine
  // Zeichenkette oder eine Zahl) ist die gewählte Sprache
  chooseLanguage(value: SegmentValue | undefined): void {
    this.tr.changeLanguage(String(value));
  }
}
